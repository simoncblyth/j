I got the same problem with my OJ build on the workstation::

    source /home/blyth/junosw/InstallArea/envset.sh

The problem was caused by the CSGOptiX__optixpath override you had in your script::

   [ -n "$CRASH" ] && export CSGOptiX__optixpath=$OPTICKS_PREFIX/ptx/objects-Debug/CSGOptiXOPTIX/CSGOptiX7.ptx

Overriding the default optixpath is a dangerous thing to do because it
can mean that you are using a CUDA kernel that was compiled against 
old versions of headers which will differ from the headers that 
other objects are compiled against. That is a potential cause for difficult 
to find bugs because it will work for some changes and fail for others.

Removing the override with my local OJ build allows my modified version of your
muon script to complete.

#!/bin/bash

######## Part 1 #########
#SBATCH --partition=gpu
#SBATCH --qos=normal
#SBATCH --account=junogpu
#SBATCH --ntasks=1
#SBATCH --mem-per-cpu=16000
#SBATCH --gpus=v100:1
#SBATCH --job-name=Mode1

######## Part 2 #########

path=$(realpath $BASH_SOURCE)
name=$(basename $path)
stem=${name/.sh}
fold=$(basename $(dirname $path))

tmp=/tmp/$USER/opticks
export TMP=${TMP:-$tmp}
DIR=$TMP/j/$fold/$stem
mkdir -p $DIR
cd $DIR

vv=""
vv="$vv BASH_SOURCE path name stem fold tmp TMP DIR PWD"
pvv(){ for v in $vv ; do printf "%30s : %s\n" "$v" "${!v}" ; done ; }
pvv


lscpu
nvidia-smi

source  /cvmfs/opticks.ihep.ac.cn/oj/releases/J25.4.0_Opticks-v0.4.5/el9_amd64_gcc11/Fri/envset.sh

export CUDA_VISIBLE_DEVICES=0
export OPTICKS_MAX_BOUNCE=63
export OPTICKS_EVENT_MODE=Minimal
export OPTICKS_HIT_MASK=EC  ## EC:EFFICIENCY_COLLECT changed from default SD:SURFACE_DETECT

source dbg__.sh

dbg__ \
        $(which python) \
        $(which tut_detsim.py) \
        --no-gdml \
        --pmtsd-merge-twindow 1.0 \
        --event-split \
        --output oj_edm.root \
        --user-output oj_user.root \
        --evtmax 100 \
        --seed 42 \
        hepevt \
        --exe Muon;

touch done_flag
